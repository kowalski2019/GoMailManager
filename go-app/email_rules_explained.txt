require ["variables", "redirect", "editheader"];

# Store the original subject
if header :matches "Subject" "*" {
  set "original_subject" "${1}";
}

# Create new subject with FWD prefix
if string "${original_subject}" :matches "??*" {
  # Subject already exists, add FWD prefix
  set "new_subject" "FWD: ${original_subject}";
} else {
  # No subject exists
  set "new_subject" "FWD: (No Subject)";
}

# Replace the subject with the new one
deleteheader "Subject";
addheader "Subject" "${new_subject}";

# Redirect the modified message
redirect "someone@example.com";
